function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function waitForIdle() {
  await sleep(400);

  while (
    $("#ctl00_ContentPlaceHolder1_UpdateProgress1").length &&
    $("#ctl00_ContentPlaceHolder1_UpdateProgress1").css("display") !== "none"
  ) {
    await sleep(400);
  }

  await sleep(500);
}
async function selectByValue(id, value) {
  const select = document.getElementById(id);

  if (!select) {
    console.log("SELECT NOT FOUND:", id);
    return false;
  }

  const optionExists = Array.from(select.options)
    .some(option => String(option.value) === String(value));

  if (!optionExists) {
    console.log("VALUE NOT FOUND:", value);
    return false;
  }

  select.value = value;

  console.log(
    "Selected:",
    select.options[select.selectedIndex].text,
    "| Value:",
    select.value
  );

  $(select).trigger("change");

  await waitForIdle();

  return true;
}
async function selectByPartialText(id, partialText) {
  const select = document.getElementById(id);

  if (!select) {
    console.log("SELECT NOT FOUND:", id);
    return false;
  }

  const search = partialText.trim().toLowerCase();

  for (let j = 0; j < select.options.length; j++) {
    const optionText = select.options[j].text
      .trim()
      .toLowerCase();

    if (optionText.includes(search)) {
      select.selectedIndex = j;

      console.log(
        "Selected:",
        select.options[j].text
      );

      $(select).trigger("change");

      await waitForIdle();

      return true;
    }
  }

  console.log("NOT FOUND:", partialText);
  return false;
}

async function selectFirstRealOption(id) {
  const select = document.getElementById(id);

  if (!select) {
    console.log("SELECT NOT FOUND:", id);
    return false;
  }

  for (let j = 0; j < select.options.length; j++) {
    const option = select.options[j];

    const value = String(option.value || "").trim();
    const text = String(option.text || "").trim().toLowerCase();

    if (
      value !== "" &&
      value !== "-1" &&
      text !== "--select--" &&
      text !== "select"
    ) {
      select.selectedIndex = j;

      console.log(
        "Selected first option:",
        option.text
      );

      $(select).trigger("change");

      await waitForIdle();

      return true;
    }
  }

  console.log("No valid option found:", id);
  return false;
}

async function setInput(id, value) {
  const el = document.getElementById(id);

  if (!el) {
    console.log("INPUT NOT FOUND:", id);
    return false;
  }

  $(el)
    .val(value)
    .trigger("input")
    .trigger("change");

  await waitForIdle();

  console.log(
    "Set:",
    id,
    "=",
    value
  );

  return true;
}
async function setDateInput(id, value) {
  const el = document.getElementById(id);

  if (!el) {
    console.log("DATE INPUT NOT FOUND:", id);
    return false;
  }

  el.focus();

  $(el)
    .val(value)
    .trigger("input")
    .trigger("change")
    .trigger("blur");

  await waitForIdle();

  console.log("Date set:", value);

  return true;
}
async function setDateLast(id, value) {
  await waitForIdle();

  const el = document.getElementById(id);

  if (!el) {
    console.log("DATE INPUT NOT FOUND:", id);
    return false;
  }

  el.value = value;

  el.dispatchEvent(new Event("input", { bubbles: true }));
  el.dispatchEvent(new Event("change", { bubbles: true }));

  await sleep(300);

  console.log("Date set:", el.value);

  return true;
}
